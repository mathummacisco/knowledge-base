      reader actually hit, so neither absolute paths nor markdown-rewritten
      relative paths work for every case. The inline script below detects

      rewrites the three buttons. Falls back to "/" if the site sits at the

  <script>
    (function () {
      var path = window.location.pathname || "/";
      var m = path.match(/^\/[^\/]+\//);
      var prefix = m ? m[0] : "/";
      document.querySelectorAll("[data-kb-404-href]").forEach(function (el) {
        el.setAttribute("href", prefix + el.getAttribute("data-kb-404-href"));
      });
    })();
  </script>
